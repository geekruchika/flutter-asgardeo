Basic Authentication Flow

A complete example showing user sign-in and sign-out.

import 'package:flutter/material.dart';
import 'package:asgardeo_flutter/asgardeo_flutter.dart';

void main() {
  runApp(
    AsgardeoProvider(
      config: AsgardeoConfig(
        baseUrl: 'https://api.asgardeo.io/t/myorg',
        clientId: 'your-client-id',
        afterSignInUrl: 'myapp://callback',
        afterSignOutUrl: 'myapp://callback',
        scopes: ['openid', 'profile', 'email'],
      ),
      child: const MyApp(),
    ),
  );
}

class MyApp extends StatelessWidget {

Protected Routes

Protect specific routes by checking authentication status.

import 'package:flutter/material.dart';
import 'package:asgardeo_flutter/asgardeo_flutter.dart';

class AppRouter {
  static Route<dynamic> generateRoute(RouteSettings settings) {
    switch (settings.name) {
      case '/':
        return MaterialPageRoute(builder: (_) => const HomePage());
      case '/dashboard':
        return MaterialPageRoute(
          builder: (context) => const ProtectedRoute(
            child: DashboardPage(),
          ),
        );
      case '/profile':
        return MaterialPageRoute(
          builder: (context) => const ProtectedRoute(
            child: ProfilePage(),
          ),
        );
      default:
        return MaterialPageRoute(builder: (_) => const NotFoundPage());
    }
  }
}

class ProtectedRoute extends StatelessWidget {
  final Widget child;
  
  const ProtectedRoute({super.key, required this.child});

  @override
  Widget build(BuildContext context) {
    return SignedIn(
      fallback: Scaffold(
        body: Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              const Icon(Icons.lock, size: 48),
              const SizedBox(height: 16),
              const Text('Please sign in to access this page'),
              const SizedBox(height: 24),
              ElevatedButton(
                onPressed: () => Navigator.pushReplacementNamed(context, '/'),
                child: const Text('Go to Home'),
              ),
            ],
          ),
        ),
      ),
      child: child,
    );
  }
}

Display User Profile

Show authenticated user information.

import 'package:flutter/material.dart';
import 'package:asgardeo_flutter/asgardeo_flutter.dart';

class ProfilePage extends StatelessWidget {
  const ProfilePage({super.key});

  @override
  Widget build(BuildContext context) {
    final auth = AsgardeoProvider.of(context);

    return Scaffold(
      appBar: AppBar(
        title: const Text('My Profile'),
      ),
      body: SignedIn(
        child: FutureBuilder<User>(
          future: auth.getUserProfile(),
          builder: (context, snapshot) {
            if (snapshot.connectionState == ConnectionState.waiting) {
              return const Center(child: CircularProgressIndicator());
            }

            if (snapshot.hasError) {
              return Center(
                child: Text('Error: ${snapshot.error}'),
              );
            }

            final user = snapshot.data;
            if (user == null) {
              return const Center(child: Text('No user data'));
            }

            return ListView(
              padding: const EdgeInsets.all(16),
              children: [
                Center(
                  child: CircleAvatar(
                    radius: 50,
                    child: Text(
                      user.username?.substring(0, 1).toUpperCase() ?? '?',
                      style: const TextStyle(fontSize: 40),
                    ),
                  ),
                ),
                const SizedBox(height: 24),
                _buildInfoCard('Username', user.username ?? 'N/A'),
                _buildInfoCard('Email', user.email ?? 'N/A'),
                _buildInfoCard('User ID', user.id ?? 'N/A'),
                const SizedBox(height: 24),
                Center(
                  child: ElevatedButton.icon(
                    icon: const Icon(Icons.logout),
                    label: const Text('Sign Out'),
                    onPressed: () async {
                      await auth.signOut();
                      if (context.mounted) {
                        Navigator.pushReplacementNamed(context, '/');
                      }
                    },
                  ),
                ),
              ],
            );
          },
        ),
      ),
    );
  }

  Widget _buildInfoCard(String label, String value) {
    return Card(
      margin: const EdgeInsets.only(bottom: 12),
      child: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              label,
              style: const TextStyle(
                fontSize: 12,
                color: Colors.grey,
                fontWeight: FontWeight.w500,
              ),
            ),
            const SizedBox(height: 4),
            Text(
              value,
              style: const TextStyle(fontSize: 16),
            ),
          ],
        ),
      ),
    );
  }
}

Making Authenticated API Calls

Use access tokens to make secure API requests.

import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;
import 'dart:convert';
import 'package:asgardeo_flutter/asgardeo_flutter.dart';

class ApiService {
  final BuildContext context;

  ApiService(this.context);

  Future<Map<String, dynamic>> fetchUserData() async {
    final auth = AsgardeoProvider.of(context);
    final token = await auth.getAccessToken();

    final response = await http.get(
      Uri.parse('https://api.example.com/user/me'),
      headers: {
        'Authorization': 'Bearer $token',
        'Content-Type': 'application/json',
      },
    );

    if (response.statusCode == 200) {
      return json.decode(response.body);
    } else if (response.statusCode == 401) {
      // Token expired, try to refresh
      await auth.signOut();
      throw Exception('Authentication expired');
    } else {
      throw Exception('Failed to load data');
    }
  }

  Future<void> updateUserProfile(Map<String, dynamic> data) async {
    final auth = AsgardeoProvider.of(context);
    final token = await auth.getAccessToken();

    final response = await http.put(
      Uri.parse('https://api.example.com/user/profile'),
      headers: {
        'Authorization': 'Bearer $token',
        'Content-Type': 'application/json',
      },
      body: json.encode(data),
    );

    if (response.statusCode != 200) {
      throw Exception('Failed to update profile');
    }
  }
}

class DataPage extends StatefulWidget {
  const DataPage({super.key});

  @override
  State<DataPage> createState() => _DataPageState();
}

class _DataPageState extends State<DataPage> {
  late ApiService _apiService;
  Map<String, dynamic>? _userData;
  bool _loading = true;
  String? _error;

  @override
  void initState() {
    super.initState();
    _apiService = ApiService(context);
    _loadData();
  }

  Future<void> _loadData() async {
    setState(() {
      _loading = true;
      _error = null;
    });

    try {
      final data = await _apiService.fetchUserData();
      setState(() {
        _userData = data;
        _loading = false;
      });
    } catch (e) {
      setState(() {
        _error = e.toString();
        _loading = false;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('My Data')),
      body: SignedIn(
        child: _loading
            ? const Center(child: CircularProgressIndicator())
            : _error != null
                ? Center(
                    child: Column(
                      mainAxisAlignment: MainAxisAlignment.center,
                      children: [
                        Text('Error: $_error'),
                        ElevatedButton(
                          onPressed: _loadData,
                          child: const Text('Retry'),
                        ),
                      ],
                    ),
                  )
                : ListView(
                    padding: const EdgeInsets.all(16),
                    children: [
                      Text('Data: ${_userData.toString()}'),
                    ],
                  ),
      ),
    );
  }
}

Organization Switching

Complete implementation of organization switching in a multi-tenant app.

import 'package:flutter/material.dart';
import 'package:asgardeo_flutter/asgardeo_flutter.dart';

class OrganizationPage extends StatefulWidget {
  const OrganizationPage({super.key});

  @override
  State<OrganizationPage> createState() => _OrganizationPageState();
}

class _OrganizationPageState extends State<OrganizationPage> {
  List<Organization> _organizations = [];
  bool _loading = true;
  String? _error;

  @override
  void initState() {
    super.initState();
    _loadOrganizations();
  }

  Future<void> _loadOrganizations() async {
    setState(() {
      _loading = true;
      _error = null;
    });

    try {
      final auth = AsgardeoProvider.of(context);
      final orgs = await auth.getMyOrganizations();
      setState(() {
        _organizations = orgs;
        _loading = false;
      });
    } catch (e) {
      setState(() {
        _error = e.toString();
        _loading = false;
      });
    }
  }

  Future<void> _switchOrganization(Organization org) async {
    final auth = AsgardeoProvider.of(context);
    
    // Show loading indicator
    showDialog(
      context: context,
      barrierDismissible: false,
      builder: (context) => const Center(
        child: CircularProgressIndicator(),
      ),
    );

    try {
      await auth.switchOrganization(org);
      
      if (mounted) {
        Navigator.pop(context); // Close loading dialog
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('Switched to ${org.name ?? org.id}'),
            backgroundColor: Colors.green,
          ),
        );
      }
    } catch (e) {
      if (mounted) {
        Navigator.pop(context); // Close loading dialog
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('Failed to switch: $e'),
            backgroundColor: Colors.red,
          ),
        );
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final auth = AsgardeoProvider.of(context);
    final currentOrg = auth.currentOrganization;

    return Scaffold(
      appBar: AppBar(
        title: const Text('Organizations'),
      ),
      body: SignedIn(
        child: _loading
            ? const Center(child: CircularProgressIndicator())
            : _error != null
                ? Center(
                    child: Column(
                      mainAxisAlignment: MainAxisAlignment.center,
                      children: [
                        const Icon(Icons.error, size: 48, color: Colors.red),
                        const SizedBox(height: 16),
                        Text('Error: $_error'),
                        const SizedBox(height: 16),
                        ElevatedButton(
                          onPressed: _loadOrganizations,
                          child: const Text('Retry'),
                        ),
                      ],
                    ),
                  )
                : _organizations.isEmpty
                    ? const Center(
                        child: Text('No organizations found'),
                      )
                    : ListView.builder(
                        itemCount: _organizations.length,
                        itemBuilder: (context, index) {
                          final org = _organizations[index];
                          final isCurrent = currentOrg?.id == org.id;

                          return Card(
                            margin: const EdgeInsets.symmetric(
                              horizontal: 16,
                              vertical: 8,
                            ),
                            child: ListTile(
                              leading: CircleAvatar(
                                child: Text(
                                  org.name?.substring(0, 1).toUpperCase() ?? '?',
                                ),
                              ),
                              title: Text(org.name ?? org.orgHandle ?? org.id),
                              subtitle: Text('ID: ${org.id}'),
                              trailing: isCurrent
                                  ? const Chip(
                                      label: Text('Current'),
                                      backgroundColor: Colors.green,
                                    )
                                  : TextButton(
                                      onPressed: () => _switchOrganization(org),
                                      child: const Text('Switch'),
                                    ),
                            ),
                          );
                        },
                      ),
      ),
    );
  }
}

import 'package:flutter/material.dart';

class CustomSignInPage extends StatefulWidget {
  const CustomSignInPage({super.key});

  @override
  State<CustomSignInPage> createState() => _CustomSignInPageState();
}

class _CustomSignInPageState extends State<CustomSignInPage> {
  bool _loading = false;

  Future<void> _handleSignIn() async {
    setState(() => _loading = true);

    try {
      final auth = AsgardeoProvider.of(context);
      final user = await auth.signIn();
      
      if (mounted && user != null) {
        Navigator.pushReplacementNamed(context, '/dashboard');

      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(

Comprehensive Error Handling

Handle different error scenarios gracefully.

import 'package:flutter/material.dart';
import 'dart:async';
import 'dart:io';
import 'package:asgardeo_flutter/asgardeo_flutter.dart';

class RobustAuthPage extends StatefulWidget {
  const RobustAuthPage({super.key});

  @override
  State<RobustAuthPage> createState() => _RobustAuthPageState();
}

class _RobustAuthPageState extends State<RobustAuthPage> {
  String? _errorMessage;
  bool _loading = false;

  Future<void> _handleSignIn() async {
    setState(() {
      _loading = true;
      _errorMessage = null;
    });

    try {
      final auth = AsgardeoProvider.of(context);
      await auth.signIn();
      
      if (mounted) {
        Navigator.pushReplacementNamed(context, '/dashboard');
      }
    } on TimeoutException {
      setState(() {
        _errorMessage = 'Request timed out. Please check your connection.';
      });
    } on SocketException {
      setState(() {
        _errorMessage = 'No internet connection. Please try again.';
      });
    } on HttpException catch (e) {
      setState(() {
        _errorMessage = 'Network error: ${e.message}';
      });
    } catch (e) {
      setState(() {
        _errorMessage = 'An unexpected error occurred: $e';
      });
    } finally {
      if (mounted) {
        setState(() => _loading = false);
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Sign In')),
      body: Center(
        child: Padding(
          padding: const EdgeInsets.all(24),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              if (_errorMessage != null) ...[
                Container(
                  padding: const EdgeInsets.all(16),
                  decoration: BoxDecoration(
                    color: Colors.red.shade50,
                    borderRadius: BorderRadius.circular(8),
                    border: Border.all(color: Colors.red),
                  ),
                  child: Row(
                    children: [
                      const Icon(Icons.error, color: Colors.red),
                      const SizedBox(width: 12),
                      Expanded(
                        child: Text(
                          _errorMessage!,
                          style: const TextStyle(color: Colors.red),
                        ),
                      ),
                    ],
                  ),
                ),
                const SizedBox(height: 24),
              ],
              ElevatedButton(
                onPressed: _loading ? null : _handleSignIn,
                child: _loading
                    ? const SizedBox(
                        width: 20,
                        height: 20,
                        child: CircularProgressIndicator(strokeWidth: 2),
                      )
                    : const Text('Sign In'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

Tips & Best Practices

1. Always Handle Deep Links

Make sure to set up deep link handling in your app's StatefulWidget to complete the auth flow.

2. Use Try-Catch Blocks

Wrap all authentication operations in try-catch blocks to handle errors gracefully.

3. Secure Token Storage

In production, use secure storage solutions like flutter_secure_storage for tokens.

4. Test on Real Devices

Deep linking and browser authentication work differently on simulators vs real devices. Always test on physical devices.

5. Handle Token Refresh

The SDK automatically refreshes expired tokens when you call getAccessToken().
